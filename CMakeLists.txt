cmake_minimum_required(VERSION 3.22)
project(quest_displays VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client>=1.20)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core DBus)
find_package(Qt5 5.15 REQUIRED COMPONENTS Gui Widgets)
find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
set(PROTOCOL ${CMAKE_CURRENT_SOURCE_DIR}/protocols/zkde-screencast-unstable-v1.xml)
set(GENERATED ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${GENERATED})
add_custom_command(OUTPUT ${GENERATED}/screencast-client.h ${GENERATED}/screencast-protocol.c
    COMMAND ${WAYLAND_SCANNER} client-header ${PROTOCOL} ${GENERATED}/screencast-client.h
    COMMAND ${WAYLAND_SCANNER} private-code ${PROTOCOL} ${GENERATED}/screencast-protocol.c
    DEPENDS ${PROTOCOL} VERBATIM)
add_executable(quest-displays host/virtual_display/main.cpp
    ${GENERATED}/screencast-client.h ${GENERATED}/screencast-protocol.c)
target_include_directories(quest-displays PRIVATE ${GENERATED})
target_link_libraries(quest-displays PRIVATE PkgConfig::WAYLAND Qt5::Core Qt5::DBus)
target_compile_options(quest-displays PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME diagnostics COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -v)
set_tests_properties(diagnostics PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME cli-help COMMAND quest-displays --help)
pkg_check_modules(WAYLAND_SERVER REQUIRED IMPORTED_TARGET wayland-server)
add_custom_command(OUTPUT ${GENERATED}/screencast-server.h
    COMMAND ${WAYLAND_SCANNER} server-header ${PROTOCOL} ${GENERATED}/screencast-server.h
    DEPENDS ${PROTOCOL} VERBATIM)
add_executable(protocol-test tests/protocol_test.cpp ${GENERATED}/screencast-server.h
    ${GENERATED}/screencast-client.h ${GENERATED}/screencast-protocol.c)
target_include_directories(protocol-test PRIVATE ${GENERATED})
target_link_libraries(protocol-test PRIVATE PkgConfig::WAYLAND PkgConfig::WAYLAND_SERVER Qt5::Core Qt5::DBus)
target_compile_options(protocol-test PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME protocol-lifetime COMMAND protocol-test)
set_tests_properties(protocol-lifetime PROPERTIES TIMEOUT 15)

# Prefer installed development packages; the project-local extraction is a
# fallback for this machine. Only headers come from .deps, not media runtimes.
pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0 gobject-2.0)
option(QUEST_BUILD_MEDIA "Build capture, NVENC and transport" ON)
if(QUEST_BUILD_MEDIA)
    find_path(GST_INCLUDE gst/gst.h PATHS /usr/include/gstreamer-1.0
        ${CMAKE_SOURCE_DIR}/.deps/root/usr/include/gstreamer-1.0 REQUIRED)
    foreach(lib gstreamer gstapp gstbase gstvideo)
        find_library(${lib}_LIB NAMES ${lib}-1.0 lib${lib}-1.0.so.0 REQUIRED)
    endforeach()
    add_library(quest-capture host/capture/capture.cpp)
    target_include_directories(quest-capture PUBLIC host ${GST_INCLUDE})
    target_link_libraries(quest-capture PUBLIC Qt5::Core Qt5::Gui PkgConfig::GLIB
        ${gstreamer_LIB} ${gstapp_LIB} ${gstbase_LIB} ${gstvideo_LIB})
    target_compile_options(quest-capture PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(quest-capture-test host/capture/capture_test.cpp)
    target_link_libraries(quest-capture-test PRIVATE quest-capture)
    pkg_check_modules(AVCODEC REQUIRED IMPORTED_TARGET libavcodec libavutil)
    add_library(quest-encoder host/encoder/nvenc.cpp)
    target_link_libraries(quest-encoder PUBLIC quest-capture PkgConfig::AVCODEC)
    target_compile_options(quest-encoder PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(quest-streams host/stream_main.cpp)
    add_library(quest-transport host/transport/tcp_server.cpp)
    target_link_libraries(quest-transport PUBLIC quest-encoder)
    target_compile_options(quest-transport PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(quest-streams PRIVATE quest-transport)
    target_compile_options(quest-streams PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(transport-test tests/transport_test.cpp)
    target_link_libraries(transport-test PRIVATE quest-transport)
    add_test(NAME transport-backpressure COMMAND transport-test)
    set_tests_properties(transport-backpressure PROPERTIES TIMEOUT 10)
endif()
add_executable(quest-test-pattern host/diagnostics/test_pattern.cpp)
target_link_libraries(quest-test-pattern PRIVATE Qt5::Widgets)
